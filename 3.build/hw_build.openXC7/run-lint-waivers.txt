# Accepted lint findings, read by run-lint.sh.
#
# One waiver per line:  <phase> <regex>
#   <phase>  tags | verilator | yosys | reset | slang | verible | ghdl | ghdl-verilator
#   <regex>  extended regex (grep -E), matched against one finding line as it
#            appears in <phase>.all of a run-lint.sh work directory
# A finding that matches is waived. Put the reason on a comment line above it.
#
# Guideline: resolve a finding at its root; waive only as the last resort.
# Work down this list and stop at the first step that applies:
#   1. Dead logic in our RTL: delete it (unused signals, registers, ports),
#      and prove the change with run-leq.sh.
#   2. Oversized buses in our RTL: size them to what is used.
#   3. Unused outputs of primitives and third-party cores, which we cannot
#      change:
#        a single pin             .PIN (/*LINT:OK*/)
#        part of a bus pin        .PIN ({NC_name, used})
#      NC_ means no-connect: driven by such an output and never read. Do not
#      give an NC_ name to anything else.
#   4. Unused inputs: tie them off.
#   5. Only then a waiver in this file: for findings with no source line of
#      ours (generated or third-party code) or a proven tool false positive.
#      Each one names the file and the signals (never a whole file or warning
#      class), and has its reason on a comment line above.
#
# The inline forms are checked: the tags phase flags a /*LINT:OK*/ that is
# not on an output.

# No-connect signals (step 3): unused by design, so every NC_ name is waived
verilator ^%Warning-UNUSEDSIGNAL: .*'NC_[A-Za-z0-9_]+'

# IMEM takes the full CPU word address and decodes only its own depth, so it
# is mirrored across the instruction space (soc_cpu.PICORV32.sv: _decode)
verilator ^%Warning-UNUSEDSIGNAL: 1\.hw/ip\.cpu/imem\.sv:[0-9]+:[0-9]+: Bits of signal are not used: '(raddr|waddr)'\[31:[0-9]+\]$

# PeakRDL regblock output (generated, rewritten by MakefileCSR), as on openPCIE:
# - field_combo / field_storage are unpacked structs whose members are each
#   written by their own always_comb / always_ff. Legal SV (one longest static
#   prefix per process); Verilator flags the whole variable:
#   https://github.com/verilator/verilator/issues/4226
# - unnamed generate blocks, an address decode bit and package constants the
#   design does not use
# - error-response logic, only read with --err-if-bad-addr / --err-if-bad-rw,
#   which MakefileCSR does not use
verilator ^%Warning-UNUSEDSIGNAL: bld/csr_build/generated-files/csr\.sv:[0-9]+:[0-9]+: Signal is not used: '(decoded_err|is_valid_addr|is_valid_rw)'$
verilator ^%Warning-MULTIDRIVEN: bld/csr_build/generated-files/csr\.sv:[0-9]+:[0-9]+: Variable written to in always_(comb|ff) also written by other process .*: '(field_combo|field_storage)'$
verilator ^%Warning-GENUNNAMED: bld/csr_build/generated-files/csr\.sv:[0-9]+:[0-9]+: Unnamed generate block 'genblk[0-9]+'
verilator ^%Warning-UNUSEDSIGNAL: bld/csr_build/generated-files/csr\.sv:[0-9]+:[0-9]+: Bits of signal are not used: 'decoded_addr'
verilator ^%Warning-UNUSEDPARAM: bld/csr_build/generated-files/csr_pkg\.sv:[0-9]+:[0-9]+: Parameter is not used: '(CSR_DATA_WIDTH|CSR_SIZE)'$
# - the cryptokey table ends at the top of the 14-bit CSR address space
#   (0x2000 + 0x1fff, entries padded to 32 words): its "addr <= end" decode
#   compare is always true (PeakRDL-regblock#225, tracked in #52)
verilator ^%Warning-CMPCONST: bld/csr_build/generated-files/csr\.sv:[0-9]+:[0-9]+: Comparison is constant due to limited range$

# Simulation view (VIEW=sim): the PLL wrappers' SIM_ONLY models are free-running
# behavioural clocks; the chip view uses clk and rst_n on PLLE2_BASE
verilator ^%Warning-UNUSEDSIGNAL: 1\.hw/fpgatech_lib/XILINX/fpga_pll_(80M|125M)\.sv:[0-9]+:[0-9]+: Signal is not used: '(clk|rst_n)'$

# Our RTL rule (1.hw/README.md): every unique case has a default, also a full
# one (in eduBOS5 an illegal opcode halts, an unknown store width writes
# nothing): unique checks that the items never overlap, the default that no
# value goes unhandled. slang -Wextra flags it (-Wcase-redundant-default)
slang ^1\.hw/[^ ]+\.sv:[0-9]+:[0-9]+: warning: .case[zx]?. marked .(unique|priority). has .default. label

# The PicoRV32 pcpi units (third party) are Verible-formatted like our own RTL
# but still Verilog 2001: untyped parameters, unlabelled generate blocks, [N]
# unpacked dimensions, untyped function arguments, always @*. Waived for these
# files only
verible ^1\.hw/ip\.cpu/cpu\.PICORV32/picorv32_pcpi_[a-z_]+\.sv:[0-9]+:[0-9-]+: .*\[(explicit-parameter-storage-type|unpacked-dimensions-range-ordering|generate-label|generate-label-prefix|explicit-function-task-parameter-type)\]$
verible ^1\.hw/ip\.cpu/cpu\.PICORV32/picorv32_pcpi_[a-z_]+\.sv:[0-9]+:[0-9-]+: .*\[always-comb\]$

# reset: data and address flops need no reset (our rule: control flops and
# flags do). These reach a select through a compare or an address decode, each
# anchored to its instance path.
#   eduBOS5: operands, load/store offset, data address, ALU result and operand
#   signs, the RV32M divider's data path; IMEM: the instruction word
reset 'u_cpu\.u_cpu\.(rs1_reg|rs2_reg|ls_offset_reg|dmem_addr|g_md\.(div_res|dvs|r_sh)|u_alu\.(alu_minus_reg|aluin_[12]_31_reg))'
reset 'u_cpu\.u_imem\.rdat'
# eduBOS5 checked alone (USE_HW_MUL=1, EN_COUNTERS=1), as on openPCIE
reset '(rs1_reg|rs2_reg|ls_offset_reg|dmem_addr|g_md\.(div_res|dvs|r_sh))'
reset 'u_alu\.(alu_minus_reg|aluin_[12]_31_reg)'
#   MAC: the GMII receive byte pipeline (preamble, SFD compares)
reset 'u_eth_[1-4]\.eth\.eth_mac_1g_gmii_inst\.eth_mac_1g_inst\.axis_gmii_rx_inst\.gmii_rxd_d[0-4]'
#   ssio_sdr_in: the IOB register that samples a pin every clock (no feedback;
#   a reset would keep the register out of the IOB)
reset 'u_eth_[1-4]\.eth\.eth_mac_1g_gmii_inst\.gmii_phy_if_inst\.rx_ssio_sdr_inst\.output_q_reg'
#   DPE input pipeline: tdata (header fields), as a block top and in the
#   switch; and the DPE input skid buffer's tdata the disassembler parses.
#   Each is read only with its tvalid, which is reset
reset '(u_dpe\.u_dpe_switch\.u_(en|de)cryptor\.)?u_dpe_if_pipeline_input\.pipeline\.g_pipe_reg\[0\]\.reg_inst\.g_skid\.m_axis_tdata_reg'
reset 'u_dpe\.u_dpe_mux\.u_dpe_if_skid_buffer_to_dpe\.sbuff\.g_skid\.m_axis_tdata_reg'
#   The register slice on the shared core's decrypt output: tdata (the
#   plaintext's first beat, its IPv4 length), read only with its tvalid
reset 'u_dpe\.u_dpe_switch\.u_dec_out_reg\.g_skid\.m_axis_tdata_reg'

# external_lib: a reset synchronizer's output resets the async FIFO reset
# synchronizers (async) and the logic of its clock domain (sync), by design
verilator ^%Warning-SYNCASYNCNET: 1\.hw/external_lib/(axis/sync_reset|ethernet/gmii_phy_if)\.sv:[0-9]+:[0-9]+: Signal flopped as both synchronous and async: '[^']*\.(sync_reg|tx_rst_reg|rx_rst_reg)'$
# external_lib: an axis_fifo built with ID_ENABLE = 0 (the MAC FIFOs) reads no tid
verilator ^%Warning-UNUSEDSIGNAL: 1\.hw/external_lib/axis/axis_fifo\.sv:[0-9]+:[0-9]+: Signal is not used: 's_axis_tid'$
# external_lib: the receive tuser (bad frame) of the MAC FIFO is not used
verilator ^%Warning-UNUSEDSIGNAL: 1\.hw/external_lib/ethernet/eth_mac_1g_gmii_fifo\.sv:[0-9]+:[0-9]+: Signal is not used: 'rx_axis_tuser'$
# CPU=picorv32: read data (PicoRV32's mem_rdata_q, the data bus, the
# routing and cryptokey table RAMs) reaches the instruction decode, a select
reset 'u_cpu\.u_cpu\.mem_rdata_q'
reset 'bus_dmem\.rdat'
reset 'u_dpe\.u_dpe_switch\.u_(cryptokey|routing)_table\.dout_a'
# CPU=picorv32 with RV32M: the multiplier's and divider's operands
reset 'u_cpu\.u_cpu\.(g_enable_mul\.pcpi_mul\.rs1|g_enable_div\.pcpi_div\.(dividend|divisor))'

# cdc, external_lib axis_async_fifo (frame FIFO): the committed write pointer
# crosses as a bus held stable by the wr_ptr_update toggle and its ack
# (sampled on the synchronized toggle edge, written again only after the ack);
# a CE-controlled crossing is that handshake, not a missing synchronizer.
# set_bus_skew in top.xdc bounds the bus against the toggle
cdc CDC-15 .* u_eth_[1-4]/eth/(rx|tx)_fifo/fifo_inst/wr_ptr_sync_commit_reg_reg\[[0-9]+\]/C +u_eth_[1-4]/eth/(rx|tx)_fifo/fifo_inst/wr_ptr_commit_sync_reg_reg\[[0-9]+\]/D$
# cdc, external_lib axis_async_fifo: the read pointer crosses gray-coded (one
# bit changes per clock) into a 2-stage ASYNC_REG synchronizer; Vivado cannot
# see the gray code. set_bus_skew in top.xdc keeps the skew under one period
cdc CDC-6 .* u_eth_[1-4]/eth/(rx|tx)_fifo/fifo_inst/rd_ptr_gray_reg_reg\[[0-9:]+\]/C +u_eth_[1-4]/eth/(rx|tx)_fifo/fifo_inst/rd_ptr_gray_sync1_reg_reg\[[0-9:]+\]/D$
